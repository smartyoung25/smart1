#### 4-B. `engine/what_if_simulator.py` 범위 동적화

```python
# 현재 (하드코딩 경계)
_BOUNDS = {"temp_internal": (10.0, 40.0), ...}

# 교체 후 (env_stats.json 기반 작물별 최적 범위)
def get_bounds(crop: str) -> dict:
    env_stats = stats_loader._load("env_stats.json")
    return env_stats[crop]["최적_범위"]

# 현재 (고정 조정 델타)
_ADJUSTMENT_DELTAS = {"temp_internal": [-2.0, -1.0, +1.0, +2.0], ...}

# 교체 후 (작물별 P25~P75 범위 기반 자동 계산)
def get_adjustment_deltas(crop: str, current_val: float) -> list:
    bounds = get_bounds(crop)[variable]
    step = (bounds[1] - bounds[0]) / 4
    return [-2*step, -step, +step, +2*step]
```
